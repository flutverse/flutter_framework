# Template files inside every leaf folder

$files = @(
    "01.Simple_Code.dart",
    "02.Dictionary.docx",
    "03.Full_Demo.dart"
)

# Loop to generate directories and files

foreach ($path in $structure) {
    # Normalize path separators for cross-platform PowerShell compatibility
    $normalizedPath = $path -replace '[/\\]', [System.IO.Path]::DirectorySeparatorChar

    # Create folder if it doesn't exist
    if (-not (Test-Path -Path $normalizedPath)) {
        New-Item -ItemType Directory -Path $normalizedPath -Force | Out-Null
    }

    # Create target template files inside folder
    foreach ($file in $files) {
        $filePath = [System.IO.Path]::Combine($normalizedPath, $file)
        if (-not (Test-Path -Path $filePath)) {
            New-Item -ItemType File -Path $filePath -Force | Out-Null
        }
    }
}

Write-Host "`nStructure '03.AdaptApp' (01 to 07) generated successfully!" -ForegroundColor Green
